A checked withdrawal
Walkthrough

Formal Silk proves the numeric contract at compile time. Parse CLI input and validate it before this call.

#require balance >= 0;
#require amount >= 0 && amount <= balance;
#assure result == balance - amount;
#assure result >= 0;
fn withdraw (balance: int, amount: int) -> int {
  return balance - amount;
}

fn main () -> int {
  let remaining = withdraw(100, 30);
  #assert remaining == 70;
  return 0;
}
