Silk

High performance language with formal verification built in.

Silk is a high performance general purpose programming language with formal verification built in. Silk targets computer systems, mobile / tablet devices, WASM / WASI runtimes, and the web.

package hello;

import { println } from "std/io";
import result from "std/result";

interface Greeter {
  fn write(name: string, score: int) -> void;
}

struct ConsoleGreeter { label: string, }

impl ConsoleGreeter as Greeter {
  fn write (self: &ConsoleGreeter, name: string, score: int) -> void {
    println("[{s}] hello {s}; score={d}", self.label, name, score);
  }
}

struct App { greeter: ConsoleGreeter, }

impl App {
  fn name (self: &App, raw: string?) -> result::Result(string, string) {
    let value: string = raw ?? "world";
    if value == "" { return Err("missing name"); }
    return Ok(value);
  }

  fn run (self: &App, raw: string?, score: int) -> int {
    let name: string = match (self.name(raw)) {
      Ok(v) => v,
      Err(_) => "guest",
    };
    self.greeter.write(name, score);
    return 0;
  }
}

fn platform () -> string {
  if attr(target="wasm32-wasi") { return "wasi"; }
  if attr(os="linux") { return "linux"; }
  return "host";
}

async fn configured_name () -> string? { return Some("silk"); }
task fn compute_score () -> int { return 21 * 2; }

async fn main () -> int {
  let app = App{ greeter: ConsoleGreeter{ label: platform() } };
  let name: string? = await configured_name();
  let score_task = compute_score();

  task {
    let score: int = yield score_task;
    return app.run(name, score);
  }
}

silk test hello.slk
silk build hello.slk -o build/hello
